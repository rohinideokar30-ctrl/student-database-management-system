#include <iostream>
using namespace std;

// Structure definition
struct Student
{
    int rollNo;
    string name;
    float marks;
};

int main()
{
    Student s[100], temp;
    int n;

    // Input number of students
    cout << "Enter Number of Students: ";
    cin >> n;

    // Add Student Records
    cout << "\nEnter Student Details\n";

    for(int i = 0; i < n; i++)
    {
        cout << "\nStudent " << i + 1 << endl;

        cout << "Enter Roll Number : ";
        cin >> s[i].rollNo;

        cout << "Enter Name        : ";
        cin >> s[i].name;

        cout << "Enter Marks       : ";
        cin >> s[i].marks;
    }

    // Display Student Records
    cout << "\n========== STUDENT DATABASE ==========\n";

    for(int i = 0; i < n; i++)
    {
        cout << "\nStudent " << i + 1 << endl;
        cout << "Roll Number : " << s[i].rollNo << endl;
        cout << "Name        : " << s[i].name << endl;
        cout << "Marks       : " << s[i].marks << endl;
    }

    // Search Student Record
    int roll;
    bool found = false;

    cout << "\nEnter Roll Number to Search: ";
    cin >> roll;

    for(int i = 0; i < n; i++)
    {
        if(s[i].rollNo == roll)
        {
            cout << "\nStudent Found";
            cout << "\nRoll Number : " << s[i].rollNo;
            cout << "\nName        : " << s[i].name;
            cout << "\nMarks       : " << s[i].marks << endl;

            found = true;
            break;
        }
    }

    if(found == false)
    {
        cout << "\nStudent Not Found.\n";
    }

    // Sort Student Records (Descending Order of Marks)
    for(int i = 0; i < n - 1; i++)
    {
        for(int j = 0; j < n - i - 1; j++)
        {
            if(s[j].marks < s[j + 1].marks)
            {
                temp = s[j];
                s[j] = s[j + 1];
                s[j + 1] = temp;
            }
        }
    }

    cout << "\n\nRecords Sorted Successfully.\n";

   // Display Sorted Records
    cout << "\n===== STUDENTS SORTED BY MARKS =====\n";

    for(int i = 0; i < n; i++)
    {
        cout << "\nRoll Number : " << s[i].rollNo;
        cout << "\nName        : " << s[i].name;
        cout << "\nMarks       : " << s[i].marks << endl;
    }

    // Update Student Record
    found = false;

    cout << "\nEnter Roll Number to Update: ";
    cin >> roll;

    for(int i = 0; i < n; i++)
    {
        if(s[i].rollNo == roll)
        {
            cout << "\nStudent Found";

            cout << "\nEnter New Name : ";
            cin >> s[i].name;

            cout << "Enter New Marks: ";
            cin >> s[i].marks;

            cout << "\nRecord Updated Successfully.\n";

            found = true;
            break;
        }
    }

    if(found == false)
    {
        cout << "\nStudent Not Found.\n";
    }

    // Display Updated Database
    cout << "\n===== UPDATED STUDENT DATABASE =====\n";

    for(int i = 0; i < n; i++)
    {
        cout << "\nRoll Number : " << s[i].rollNo;
        cout << "\nName        : " << s[i].name;
        cout << "\nMarks       : " << s[i].marks << endl;
    }

    return 0;
}
